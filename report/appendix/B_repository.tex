\subsection{Reproducing All Results}

\begin{enumerate}
  \item \textbf{Build firmware}
\begin{verbatim}
export PICO_SDK_PATH=/path/to/pico-sdk
cd firmware && mkdir build && cd build
cmake .. && make -j4
# Flash firmware.uf2 via USB mass storage
\end{verbatim}

  \item \textbf{Capture real traces} (requires Pico + oscilloscope)
\begin{verbatim}
python capture.py
# Output: data/set_001_pico_real/{traces,plaintexts,ciphertexts}.npy
\end{verbatim}

  \item \textbf{Run CPA on reference dataset}
\begin{verbatim}
python attacks/cpa.py --data data/set_002_reference --round last
\end{verbatim}

  \item \textbf{Run DPA on reference dataset}
\begin{verbatim}
python attacks/dpa.py --data data/set_002_reference --round last
\end{verbatim}

  \item \textbf{Generate Guessing Entropy plot}
\begin{verbatim}
python attacks/guessing_entropy.py \
  --data data/set_002_reference --byte 0 --trials 20
\end{verbatim}
\end{enumerate}

\noindent
All scripts read dataset path, attack parameters, and known key from
\texttt{meta.json}; no hardcoded paths.
Generated plots are saved to \texttt{data/<set>/plots/}.
